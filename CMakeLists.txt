cmake_minimum_required(VERSION 3.20)
project(resource_visualizer LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "Only the macOS backend is implemented currently")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(monitor_backend src/monitor/macos.cpp src/monitor/bridge.cpp)
target_include_directories(monitor_backend PUBLIC src/monitor)
target_link_libraries(monitor_backend PUBLIC "-framework IOKit" "-framework CoreFoundation")
add_executable(resource-visualizer src/visualizer/main.cpp)
target_include_directories(resource-visualizer PRIVATE src/visualizer)
target_link_libraries(resource-visualizer PRIVATE monitor_backend)
foreach(target monitor_backend resource-visualizer)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(visualizer-tests tests/cpp/visualizer_test.cpp)
  target_include_directories(visualizer-tests PRIVATE src/visualizer src/monitor/include)
  target_link_libraries(visualizer-tests PRIVATE monitor_backend)
  add_test(NAME visualizer-tests COMMAND visualizer-tests)
  add_test(NAME visualizer-help COMMAND resource-visualizer --help)
  add_test(NAME visualizer-invalid COMMAND resource-visualizer --interval nan)
  set_tests_properties(visualizer-invalid PROPERTIES WILL_FAIL TRUE)
endif()
