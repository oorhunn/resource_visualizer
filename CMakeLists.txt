cmake_minimum_required(VERSION 3.20)
project(resource_visualizer LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "Only the macOS backend is implemented currently")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(monitor_backend src/monitor/macos.cpp)
target_include_directories(monitor_backend PUBLIC src/monitor)
target_link_libraries(monitor_backend PUBLIC "-framework IOKit" "-framework CoreFoundation")
add_executable(resource-visualizer src/visualizer/main.cpp)
target_include_directories(resource-visualizer PRIVATE src/visualizer)
target_link_libraries(resource-visualizer PRIVATE monitor_backend)
# Embed the dashboard page; editing web/index.html re-runs this step on the next build.
file(READ web/index.html INDEX_HTML)
configure_file(src/web/index_html.hpp.in ${CMAKE_CURRENT_BINARY_DIR}/generated/index_html.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS web/index.html)
add_executable(resource-web src/web/main.cpp)
target_include_directories(resource-web PRIVATE src/web src/visualizer
                           ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(resource-web PRIVATE monitor_backend)
foreach(target monitor_backend resource-visualizer resource-web)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(visualizer-tests tests/cpp/visualizer_test.cpp)
  target_include_directories(visualizer-tests PRIVATE src/visualizer src/web)
  target_link_libraries(visualizer-tests PRIVATE monitor_backend)
  add_test(NAME visualizer-tests COMMAND visualizer-tests)
  add_test(NAME visualizer-help COMMAND resource-visualizer --help)
  add_test(NAME visualizer-invalid COMMAND resource-visualizer --interval nan)
  add_test(NAME web-help COMMAND resource-web --help)
  add_test(NAME web-invalid-port COMMAND resource-web --port 70000)
  set_tests_properties(visualizer-invalid web-invalid-port PROPERTIES WILL_FAIL TRUE)
endif()
